## สิ่งที่เปลี่ยนจากเอกสาร API ฉบับก่อน

- ใช้ **Node.js + Express** ชื่อ API เป็น `/api/xxx` ไฟล์อยู่ที่ `routes/xxx.js`
- อ้างอิงเลข **FR ใน Requirement v5** แทนเลข use case
- **เมนูมีโซนครัว** (`kitchen_zone`) จอครัวแยกตามโซนได้
- **ห้อง VIP ขนาดเดียว** 15 คน ยอดขั้นต่ำ 3,000฿ มัดจำ 50% สูงสุด 4 ชั่วโมง (ค่าเหล่านี้อยู่ในตั้งค่าร้าน)
- **ตัด QR รับแต้ม**: กรอกเบอร์สมาชิกตอนปิดบิลได้แต้มทันที ไม่กรอกไม่ได้แต้ม
- **คูปองกำหนดวันในสัปดาห์และยกเว้นวันหยุดราชการได้** + API วันหยุดราชการ
- ตอนดูยอดก่อนจ่าย ระบบแนะนำคูปองที่ใช้ได้ตอนนั้น
- แบ่งเป็น **4 เฟส** และมีคอลัมน์ **คนทำ** ตามตารางแบ่งงานใน Requirement v5

## ข้อตกลงที่ใช้ทุก API

- หน้าเว็บเรียกด้วย `fetch('/api/ชื่อapi')` ไฟล์ใน `routes/` ถูกโหลดอัตโนมัติ ไม่ต้องแก้ `server.js`
- อ่านข้อมูล = `GET` ส่งค่าทาง URL เช่น `?q=ฮะ`
- เพิ่ม แก้ ลบ และอื่นๆ = `POST` ส่ง JSON ที่มี `action` เช่น `{"action":"create", ...}` (ถ้ามีไฟล์รูปใช้ form-data)
- คำตอบทุก API: `{"success": true/false, "data": ..., "message": ""}`
- ชื่อฟิลด์ใน JSON ตรงกับชื่อในฐานข้อมูลเป๊ะ (snake_case)
- พนักงานล็อกอินด้วย session / สมาชิกใช้ session แยก / ลูกค้า walk-in ใช้ `token` จาก QR บิล
- หน้าที่ต้องอัปเดตเอง (จอครัว, ผังโต๊ะ, สถานะของลูกค้า) เรียก GET ซ้ำทุก 5–10 วินาที ไม่ต้องทำ real-time
- ทุก API ที่เปลี่ยนข้อมูลสำคัญ เรียก `writeLog()`

**ใครเรียกได้**: สาธารณะ = ไม่ต้องล็อกอิน · พนักงาน = ทุกบทบาทที่ล็อกอิน · staff+ = พนักงานหน้าร้าน ผู้จัดการ เจ้าของร้าน · manager+ = ผู้จัดการ เจ้าของร้าน · [รับเงิน] / [จอง VIP] = staff ต้องมีช่องติ๊กนั้น (ผู้จัดการและเจ้าของร้านทำได้เสมอ)

**เฟส**: P1 ฐานระบบ · P2 สั่งอาหาร ครัว ชำระเงิน · P3 ห้อง VIP สมาชิก แต้ม คูปอง · P4 ผู้จัดการ จัดซื้อ สต็อก

SET NAMES utf8mb4;

CREATE DATABASE IF NOT EXISTS dimsum
  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE dimsum;

CREATE TABLE roles (
  id          INT AUTO_INCREMENT PRIMARY KEY,
  role_code   VARCHAR(20) NOT NULL UNIQUE,
  role_name   VARCHAR(50) NOT NULL
) ENGINE=InnoDB;

CREATE TABLE users (
  id              INT AUTO_INCREMENT PRIMARY KEY,
  username        VARCHAR(50)  NOT NULL UNIQUE,
  password_hash   VARCHAR(255) NOT NULL,
  full_name       VARCHAR(100) NOT NULL,
  phone           VARCHAR(20),
  role_id         INT          NOT NULL,
  can_payment     TINYINT(1)   NOT NULL DEFAULT 1,
  can_vip_booking TINYINT(1)   NOT NULL DEFAULT 1,
  is_active       TINYINT(1)   NOT NULL DEFAULT 1,
  deactivated_at  DATETIME     NULL,
  created_at      DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at      DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  FOREIGN KEY (role_id) REFERENCES roles(id)
) ENGINE=InnoDB;

CREATE TABLE shop_settings (
  id                    INT PRIMARY KEY DEFAULT 1,
  shop_name             VARCHAR(100) NOT NULL,
  phone                 VARCHAR(20),
  email                 VARCHAR(100),
  address               VARCHAR(255),
  morning_open          TIME NOT NULL DEFAULT '10:00:00',
  morning_last_order    TIME NOT NULL DEFAULT '13:30:00',
  morning_close         TIME NOT NULL DEFAULT '14:00:00',
  evening_open          TIME NOT NULL DEFAULT '16:00:00',
  evening_last_order    TIME NOT NULL DEFAULT '19:30:00',
  evening_close         TIME NOT NULL DEFAULT '20:00:00',
  max_qty_per_item      INT  NOT NULL DEFAULT 10,
  vip_min_spend         DECIMAL(10,2) NOT NULL DEFAULT 3000,
  vip_deposit_percent   INT  NOT NULL DEFAULT 50,
  vip_max_hours         INT  NOT NULL DEFAULT 4,
  CHECK (id = 1)
) ENGINE=InnoDB;

CREATE TABLE public_holidays (
  id            INT AUTO_INCREMENT PRIMARY KEY,
  holiday_date  DATE NOT NULL UNIQUE,
  name          VARCHAR(100) NOT NULL
) ENGINE=InnoDB;

CREATE TABLE banners (
  id           INT AUTO_INCREMENT PRIMARY KEY,
  title        VARCHAR(100) NULL,
  image_path   VARCHAR(255) NOT NULL,
  link_url     VARCHAR(255) NULL,
  sort_order   INT NOT NULL DEFAULT 0,
  is_active    TINYINT(1) NOT NULL DEFAULT 1,
  created_by   INT NULL,
  created_at   DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at   DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  FOREIGN KEY (created_by) REFERENCES users(id)
) ENGINE=InnoDB;

CREATE TABLE menu_categories (
  id          INT AUTO_INCREMENT PRIMARY KEY,
  name        VARCHAR(100) NOT NULL,
  sort_order  INT NOT NULL DEFAULT 0,
  is_active   TINYINT(1) NOT NULL DEFAULT 1
) ENGINE=InnoDB;

CREATE TABLE menu_items (

  category_id    INT NOT NULL,
  name           VARCHAR(100) NOT NULL,
  description    VARCHAR(255),
  price          DECIMAL(10,2) NOT NULL,
  unit           VARCHAR(20) NOT NULL DEFAULT 'เข่ง',
  kitchen_zone   ENUM('steam','fry','congee_noodle','drink_dessert') NOT NULL DEFAULT 'steam',
  image_path     VARCHAR(255),
  lot_size       INT NOT NULL DEFAULT 1,
  available_qty  INT NULL,
  is_available   TINYINT(1) NOT NULL DEFAULT 1,
  is_deleted     TINYINT(1) NOT NULL DEFAULT 0,
  created_at     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

const mysql = require('mysql2/promise');
const config = require('../config');

// การเชื่อมต่อฐานข้อมูลชุดเดียวที่ทุกไฟล์ใช้ร่วมกัน
const pool = mysql.createPool({
  ...config.db,
  charset: 'utf8mb4',
  timezone: '+07:00',
  dateStrings: true,          // วันเวลาได้เป็นข้อความ 'YYYY-MM-DD HH:MM:SS' ตรงตามข้อตกลง
  decimalNumbers: true,       // DECIMAL ได้เป็นตัวเลข ไม่ใช่ข้อความ

/** รัน SQL แล้วคืนแถวทั้งหมด: const rows = await query('SELECT ... WHERE id = ?', [id]) */
async function query(sql, params = []) {
  const [rows] = await pool.execute(sql, params);
  return rows;
}

/** รัน SQL แล้วคืนแถวแรก หรือ null */
async function queryOne(sql, params = []) {
  const rows = await query(sql, params);
  return rows[0] || null;
}

/**
 * ทำหลายคำสั่งแบบทั้งหมดหรือไม่ทำเลย เช่น เปิดบิล + เปลี่ยนสถานะโต๊ะ
 * await transaction(async (conn) => { await conn.execute(...); await conn.execute(...); })
 */
async function transaction(work) {
  const conn = await pool.getConnection();
  try {
    await conn.beginTransaction();
    const result = await work(conn);
    await conn.commit();
    return result;
  } catch (err) {
    await conn.rollback();
    throw err;
  } finally {
    conn.release();
  }
}

module.exports = { pool, query, queryOne, transaction };

// รูปแบบคำตอบกลางของทุก API: {"success": true/false, "data": ..., "message": ""}

class ApiError extends Error {
  constructor(message, status = 400, data = null) {
    super(message);
    this.status = status;
    this.data = data;
  }
}

/** ตอบสำเร็จ: return ok(res, data, 'ข้อความ') */
function ok(res, data = null, message = '') {
  return res.json({ success: true, data, message });
}

/** ตอบไม่สำเร็จ: throw fail('ข้อความ', 403) — ใช้ throw เพื่อหยุดทำงานทันที */
function fail(message, status = 400, data = null) {
  return new ApiError(message, status, data);
}

/** ตรวจว่าส่งข้อมูลครบ ไม่ครบโยน error 422 ทันที: requireFields(req.body, ['name','price']) */
function requireFields(body, fields) {
  const missing = fields.filter((f) => body[f] === undefined || body[f] === null || body[f] === '');
  if (missing.length) throw fail('ข้อมูลไม่ครบ: ' + missing.join(', '), 422);
  return body;
}

/** ห่อ handler แบบ async ให้ error ทุกชนิดไปถึงตัวจัดการ error กลาง */
const handle = (fn) => (req, res, next) => Promise.resolve(fn(req, res, next)).catch(next);

/** ตัวจัดการ error กลาง (อยู่ท้ายสุดใน server.js) ตอบเป็น JSON เสมอ */
function errorHandler(err, req, res, next) {
  if (err instanceof ApiError) {
    return res.status(err.status).json({ success: false, data: err.data, message: err.message });
  }
  console.error(err);
  return res.status(500).json({ success: false, data: null, message: 'เกิดข้อผิดพลาดในระบบ: ' + err.message });
}

module.exports = { ApiError, ok, fail, requireFields, handle, errorHandler };

 * โหลดผู้ใช้จากฐานข้อมูลทุก request (เก็บไว้ที่ req.user)
 * ถ้า Owner ปิดบัญชีหรือเปลี่ยนสิทธิ์ระหว่างที่ล็อกอินอยู่ จะมีผลทันที

async function loadUser(req) {
  if (req._userLoaded) return req.user;
  req._userLoaded = true;
  req.user = null;
  const id = req.session && req.session.userId;
  if (!id) return null;
  const u = await queryOne(
    `SELECT u.id, u.username, u.full_name, u.phone, u.role_id, r.role_code, r.role_name,
            u.can_payment, u.can_vip_booking, u.is_active
     FROM users u JOIN roles r ON r.id = u.role_id WHERE u.id = ?`, [id]);
  if (u && u.is_active === 1) req.user = u;
  else delete req.session.userId;
  return req.user;
}

function loginUser(req, userId) {

      if (err) return reject(err);
      req.session.userId = userId;
      req._userLoaded = false;
      resolve();
    }));
}

/**
 * ช่องติ๊กรายคน: 'payment' (รับเงิน) หรือ 'vip_booking' (จองห้อง VIP)
 * ผู้จัดการและเจ้าของร้านผ่านเสมอ / พนักงานหน้าร้านดูช่องติ๊ก / หัวหน้าเชฟไม่ได้
 */
function hasPerm(user, perm) {
  if (!user) return false;
  if (['manager', 'owner'].includes(user.role_code)) return true;
  if (user.role_code !== 'staff') return false;
  if (perm === 'payment') return user.can_payment === 1;
  if (perm === 'vip_booking') return user.can_vip_booking === 1;
  return false;
}

/* ----- middleware: ใส่หน้า handler ได้เลย เช่น router.get('/', requireLogin, handle(...)) ----- */

async function requireLogin(req, res, next) {
  try {
    if (!(await loadUser(req))) throw fail('กรุณาล็อกอิน', 401);
    next();
  } catch (e) { next(e); }
}

/** ต้องเป็นบทบาทที่กำหนด: requireRole(['manager','owner']) */
function requireRole(roleCodes) {
  return async (req, res, next) => {
    try {
      const u = await loadUser(req);
      if (!u) throw fail('กรุณาล็อกอิน', 401);
      if (!roleCodes.includes(u.role_code)) throw fail('ไม่มีสิทธิ์ใช้งานส่วนนี้', 403);
      next();
    } catch (e) { next(e); }
  };
}

/** ต้องมีสิทธิ์ช่องติ๊ก: requirePerm('payment') */
function requirePerm(perm) {
  return async (req, res, next) => {
    try {
      const u = await loadUser(req);
      if (!u) throw fail('กรุณาล็อกอิน', 401);
      if (!hasPerm(u, perm)) throw fail('ไม่มีสิทธิ์ทำรายการนี้', 403);
      next();
    } catch (e) { next(e); }
  };
}

/**
 * ใช้ข้างใน handler เมื่อ action ต่างกันต้องการสิทธิ์ต่างกัน (POST แยก action)
 * await mustBe(req, ['owner'])   /   await mustHavePerm(req, 'payment')
 */
async function mustBe(req, roleCodes) {
  const u = await loadUser(req);
  if (!u) throw fail('กรุณาล็อกอิน', 401);
  if (!roleCodes.includes(u.role_code)) throw fail('ไม่มีสิทธิ์ใช้งานส่วนนี้', 403);
  return u;
}

async function mustHavePerm(req, perm) {
  const u = await loadUser(req);
  if (!u) throw fail('กรุณาล็อกอิน', 401);
  if (!hasPerm(u, perm)) throw fail('ไม่มีสิทธิ์ทำรายการนี้', 403);
  return u;
}

/* ========== สมาชิก (session แยกจากพนักงาน) ========== */

function loginMember(req, memberId) { req.session.memberId = memberId; }
function logoutMember(req) { delete req.session.memberId; }

async function requireMember(req, res, next) {
  try {
    const id = req.session && req.session.memberId;
    const m = id ? await queryOne('SELECT * FROM members WHERE id = ?', [id]) : null;
    if (!m) throw fail('กรุณาล็อกอินสมาชิก', 401);
    req.member = m;
    next();
  } catch (e) { next(e); }
}

  requireLogin, requireRole, requirePerm, mustBe, mustHavePerm,
  loginMember, logoutMember, requireMember,

  }).format(new Date());

/** กะปัจจุบัน 'morning' / 'evening' หรือ null ถ้าร้านปิด */

  const s = await shopSettings();
  if (time >= s.morning_open && time < s.morning_close) return 'morning';
  if (time >= s.evening_open && time < s.evening_close) return 'evening';
  return null;
}

 * POST {"action":"login","username":"...","password":"..."}
 * POST {"action":"logout"}
 * GET  → ข้อมูลผู้ใช้ที่ล็อกอินอยู่ (ใช้สร้าง sidebar ตามบทบาท)
 */
const router = require('express').Router();
const bcrypt = require('bcryptjs');
const { queryOne } = require('../lib/db');
const { ok, fail, requireFields, handle } = require('../lib/response');
const { requireLogin, loginUser, logoutUser, loadUser } = require('../lib/auth');
const { writeLog } = require('../lib/log');

router.get('/', requireLogin, handle(async (req, res) => ok(res, req.user)));

router.post('/', handle(async (req, res) => {
  const b = req.body;
  switch (b.action) {
    case 'login': {
      requireFields(b, ['username', 'password']);
      const u = await queryOne('SELECT id, password_hash, is_active FROM users WHERE username = ?',
        [String(b.username).trim()]);
      if (!u || !(await bcrypt.compare(String(b.password), u.password_hash))) {
        throw fail('ชื่อล็อกอินหรือรหัสผ่านไม่ถูกต้อง', 401);

      if (u.is_active !== 1) throw fail('บัญชีนี้ถูกปิดการใช้งาน', 403);
      await loginUser(req, u.id);
      await writeLog(req, 'user.login', 'users', u.id);
      return ok(res, await loadUser(req), 'ล็อกอินสำเร็จ');
    }
    case 'logout': {
      if (await loadUser(req)) await writeLog(req, 'user.logout', 'users', req.user.id);
      logoutUser(req);
      return ok(res, null, 'ออกจากระบบแล้ว');
    }
    default:
      throw fail('ไม่รู้จัก action นี้', 400);

}));

/* =========================================================
   api.js กลาง — ทุกหน้าใช้ไฟล์นี้ ห้ามเขียน fetch เองในหน้า
   ห้ามแก้เอง ถ้าต้องการฟังก์ชันเพิ่ม แจ้งคนที่ 1
   ========================================================= */

const api = {
  /** อ่านข้อมูล: const menus = await api.get('menus', { q: 'ฮะ' }) → ได้ data เลย */
  async get(name, params = {}) {
    const qs = new URLSearchParams(Object.entries(params).filter(([, v]) => v !== undefined && v !== null && v !== ''));
    return api._send('/api/' + name + (qs.toString() ? '?' + qs : ''), { method: 'GET' });
  },
  /** เปลี่ยนข้อมูล: await api.post('bills', { action: 'open', table_ids: [5], guests: 4 }) */
  async post(name, body = {}) {
    return api._send('/api/' + name, {
      method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
    });
  },
  /** ส่งไฟล์รูป: const fd = new FormData(form); fd.append('action','create'); await api.upload('menus', fd) */
  async upload(name, formData) {
    return api._send('/api/' + name, { method: 'POST', body: formData });
  },
  async _send(url, opts) {
    let json;
    try {
      const res = await fetch(url, opts);
      json = await res.json();
      if (res.status === 401 && location.pathname.startsWith('/admin/') && !location.pathname.endsWith('login.html')) {
        location.href = '/admin/login.html';
      }
    } catch (e) {
      json = { success: false, data: null, message: 'เชื่อมต่อเซิร์ฟเวอร์ไม่ได้' };
    }
    if (!json.success) {
      const err = new Error(json.message || 'เกิดข้อผิดพลาด');
      err.data = json.data;
      throw err;
    }
    api.lastMessage = json.message;
    return json.data;
  },
};

/** เรียก API แล้วแสดง toast ให้อัตโนมัติ: await run(() => api.post(...), 'บันทึกแล้ว') → คืน data หรือ null ถ้า error */
async function run(fn, successText) {
  try {
    const data = await fn();
    const msg = successText || api.lastMessage;
    if (msg) toast(msg);
    return data;
  } catch (e) {
    toast(e.message, 'error');
    return null;
  }
}

/* ---------- toast ---------- */
function toast(message, type = 'ok') {
  let box = document.querySelector('.toast-box');
  if (!box) { box = document.createElement('div'); box.className = 'toast-box'; document.body.appendChild(box); }
  const t = document.createElement('div');
  t.className = 'toast' + (type === 'error' ? ' toast-error' : type === 'warn' ? ' toast-warn' : '');
  t.textContent = message;
  box.appendChild(t);
  setTimeout(() => t.remove(), 3000);
}

/* ---------- modal ---------- */
/** เปิด modal: const m = openModal({ title, body: '<html>', size: 'lg', footer: '<button>...' }); m.close() */
function openModal({ title = '', body = '', footer = '', size = '' }) {
  const bg = document.createElement('div');
  bg.className = 'modal-backdrop';
  bg.innerHTML = `
    <div class="modal ${size === 'lg' ? 'modal-lg' : ''}" role="dialog">
      <div class="modal-head"><h3>${title}</h3><button class="btn btn-ghost btn-icon btn-sm" data-close>✕</button></div>
      <div class="modal-body">${body}</div>
      ${footer ? `<div class="modal-foot">${footer}</div>` : ''}
    </div>`;
  const close = () => bg.remove();
  bg.addEventListener('click', (e) => { if (e.target === bg || e.target.closest('[data-close]')) close(); });
  document.body.appendChild(bg);
  return { el: bg.querySelector('.modal'), close };
}

/** ถามยืนยัน: if (await confirmModal('ยกเลิกโต๊ะ T08?', 'ยืนยัน', true)) { ... } */
function confirmModal(message, confirmText = 'ยืนยัน', danger = false) {
  return new Promise((resolve) => {
    const m = openModal({
      title: 'ยืนยัน',
      body: `<p>${message}</p>`,
      footer: `<button class="btn btn-secondary" data-close>ยกเลิก</button>
               <button class="btn ${danger ? 'btn-danger' : ''}" data-ok>${confirmText}</button>`,
    });
    m.el.querySelector('[data-ok]').onclick = () => { m.close(); resolve(true); };
    m.el.parentElement.addEventListener('click', (e) => { if (e.target.closest('[data-close]') || e.target === m.el.parentElement) resolve(false); });
  });
}

const HOME_BY_ROLE = { staff: 'tables.html', head_chef: 'kitchen.html', manager: 'dashboard.html', owner: 'dashboard.html' };

function canSee(user, item) {
  if (!item.roles.includes(user.role_code)) return false;
  if (item.perm && user.role_code === 'staff') {
    return item.perm === 'payment' ? user.can_payment === 1 : user.can_vip_booking === 1;
  }
  return true;
}

/** สถานะร้านจาก settings → { open, text } */
function shopStatus(s) {
  const now = new Date().toTimeString().slice(0, 8);
  if (now >= s.morning_open && now < s.morning_close)
    return { open: true, text: `เปิดอยู่ · กะเช้า · ปิดรับออเดอร์ ${s.morning_last_order.slice(0, 5)}` };
  if (now >= s.evening_open && now < s.evening_close)
    return { open: true, text: `เปิดอยู่ · กะเย็น · ปิดรับออเดอร์ ${s.evening_last_order.slice(0, 5)}` };
  return { open: false, text: 'ปิดอยู่' };
}

/**
 * เรียกบนสุดของทุกหน้าพนักงาน:
 *   const me = await initStaffPage({ title: 'ผังโต๊ะ', roles: ['staff','manager','owner'], perm: 'payment' });
 * - ไม่ได้ล็อกอิน → ไปหน้า login / ไม่มีสิทธิ์ → แสดงหน้าไม่มีสิทธิ์
 * - หน้า HTML ต้องมี <div id="app"></div> และใส่เนื้อหาใน <template id="page">...</template>
 * - คืนค่า user (id, full_name, role_code, can_payment, can_vip_booking) และ window.SHOP = ตั้งค่าร้าน
 */
async function initStaffPage({ title, roles, perm } = {}) {
  let user;
  try { user = await api.get('auth'); } catch { location.href = '/admin/login.html'; return; }
  let settings = {};
  try { settings = await api.get('settings'); } catch { /* ยังไม่มี API settings ก็ใช้ได้ */ }
  window.SHOP = settings;

  const current = location.pathname.split('/').pop();
  const menuHtml = STAFF_MENU.map((g) => {
    const items = g.items.filter((it) => canSee(user, it));
    if (!items.length) return '';
    return `<div class="side-group">${g.group}</div>` + items.map((it) =>
      `<a class="side-link ${it.file === current ? 'active' : ''}" href="/admin/${it.file}">
         <span class="side-icon">${it.icon}</span>${it.label}</a>`).join('');
  }).join('');

  const st = settings.morning_open ? shopStatus(settings) : null;
  const page = document.getElementById('page');
  document.getElementById('app').innerHTML = `
    <div class="app">
      <aside class="sidebar" id="sidebar">
        <div class="sidebar-logo"><a class="logo" href="/admin/${HOME_BY_ROLE[user.role_code] || ''}">
          <span class="logo-mark">🥟</span><span class="logo-text">นึ่งสุข</span></a></div>
        ${menuHtml}
      </aside>
      <div class="main">
        <header class="topbar">
          <div class="row">
            <button class="btn btn-ghost btn-icon sidebar-toggle" onclick="document.getElementById('sidebar').classList.toggle('open')">☰</button>
            <h2>${title || ''}</h2>
          </div>
          <div class="topbar-right">
            ${st ? `<span class="badge ${st.open ? 'badge-ok' : 'badge-off'}">● ${st.text}</span>` : ''}
            <div class="user-box">
              <span class="avatar">${fmt.esc(user.full_name.charAt(0))}</span>
              <div><div class="bold small">${fmt.esc(user.full_name)}</div><div class="caption muted">${ROLES[user.role_code]}</div></div>
            </div>
            <button class="btn btn-secondary btn-sm" id="logoutBtn">ออกจากระบบ</button>
          </div>
        </header>
        <main class="content" id="content"></main>
      </div>
    </div>`;

  document.getElementById('logoutBtn').onclick = async () => {
    await api.post('auth', { action: 'logout' }).catch(() => {});
    location.href = '/admin/login.html';
  };

  const allowed = (!roles || roles.includes(user.role_code)) && (!perm || canSee(user, { roles: [user.role_code], perm }));
  const content = document.getElementById('content');
  if (!allowed) {
    content.innerHTML = `<div class="empty"><div class="empty-icon">🔒</div><h3>ไม่มีสิทธิ์เข้าถึงหน้านี้</h3>
      <p><a class="btn" href="/admin/${HOME_BY_ROLE[user.role_code]}">กลับหน้าแรก</a></p></div>`;
    return null;
  }
  if (page) content.appendChild(page.content.cloneNode(true));
  return user;
}
